p5.js
Creative Coding Patterns

이 문서는 p5.js를 사용하여 시각 예술을 만들 때 자주 사용되는 자바스크립트 문법과 디자인 패턴을 정리한 가이드입니다. 단순한 문법 설명을 넘어, 실제 작품에 바로 쓰이는 코드 패턴 위주로 구성했습니다.

01

스케치 구조와 캔버스

1-1. 반응형 풀스크린 캔버스 (Responsive)

모던 웹 환경에서 아트워크는 다양한 화면 크기에 유연하게 대응해야 합니다. windowWidth와 windowHeight 시스템 변수를 사용하여 캔버스를 꽉 채우고, windowResized() 이벤트 함수 안에서 resizeCanvas()를 호출하여 브라우저 크기 변경 시 캔버스도 함께 조절하는 패턴은 필수적입니다. 

function setup() {
  // 처음 실행 시 브라우저 크기만큼 캔버스 생성
  createCanvas(windowWidth, windowHeight);
  background(0);
}

function draw() {
  // 항상 화면의 정중앙을 (0,0)으로 사용하기 위해 translate 활용
  translate(width/2, height/2);
  
  // 화면 크기가 변해도 비율이 유지되는 원 그리기
  let size = min(width, height) * 0.5; 
  circle(0, 0, size);
}

// 브라우저 크기가 바뀔 때 자동으로 호출되는 p5 내장 함수
function windowResized() {
  resizeCanvas(windowWidth, windowHeight);
  // 리사이즈 직후 배경을 다시 그리지 않으면 이전 그림이 깨져 보일 수 있음
  background(0); 
}

1-2. 사용자 정의 함수 (Clean Code)

draw() 함수는 오케스트라의 지휘자와 같습니다. 모든 세세한 그리기 명령을 직접 내리기보다는, "배경 그려", "캐릭터 그려", "UI 그려"와 같이 의도가 드러나는 이름의 함수를 호출하는 방식으로 구조를 짜야 코드가 복잡해져도 길을 잃지 않습니다. 

function draw() {
  // 1. 배경 처리
  drawDynamicBackground();
  
  // 2. 메인 오브젝트 그리기
  // x, y 좌표를 인자로 넘겨 재사용성을 높임
  drawCharacter(width * 0.3, height / 2);
  drawCharacter(width * 0.7, height / 2);
  
  // 3. 상단 UI 표시
  drawInterface();
}

function drawDynamicBackground() {
  background(20);
  // ... 배경 패턴 로직 ...
}

function drawCharacter(x, y) {
  push(); // 스타일 격리 (이 함수 안의 설정이 밖으로 새나가지 않음)
  translate(x, y);
  circle(0, 0, 50);
  pop();
}

function drawInterface() {
  fill(255);
  text("FPS: " + floor(frameRate()), 10, 20);
}
02

컬러 시스템

2-1. HSB 모드의 직관성

디지털 아트에서 RGB(빛의 삼원색)는 직관적으로 다루기 어렵습니다. 대신 HSB(Hue:색상, Saturation:채도, Brightness:명도) 모드를 사용하세요. Hue 값만 0에서 360까지 반복문으로 돌리면 손쉽게 무지개색 그라데이션을 만들 수 있습니다. 또한 random(palette) 패턴을 사용해 미리 정의된 색 조합만 사용하도록 제한하면 세련된 디자인을 얻을 수 있습니다. 

let palette = ['#FF6138', '#FFFF9D', '#BEEB9F', '#79BD8F', '#00A388'];

function setup() {
  createCanvas(400, 400);
  // HSB 모드 설정: Hue(0~360), Saturation(0~100), Brightness(0~100), Alpha(0~1)
  colorMode(HSB, 360, 100, 100, 1);
  noStroke();
}

function draw() {
  background(0, 0, 10); // 아주 어두운 회색 (Brightness: 10)
  
  // 1. 팔레트에서 랜덤 색상 추출 패턴
  let hexColor = random(palette); 
  let c = color(hexColor); // p5 Color 객체로 변환
  
  // 2. HSB 값을 조작하여 변형 주기
  let h = hue(c);          // 원래 색상값 가져오기
  let s = saturation(c);   // 원래 채도값
  let b = brightness(c);   // 원래 명도값
  
  // 채도와 명도를 랜덤하게 흔들어 텍스처감 부여
  fill(h, s + random(-10, 10), b + random(-10, 10), 0.8);
  
  circle(random(width), random(height), random(10, 50));
}
03

그리드와 반복

3-1. 이중 루프와 좌표 격리 (Matrix Transformation)

화면을 격자(Grid)로 나누는 것은 제너레이티브 아트의 가장 기초적인 패턴입니다. for 반복문을 중첩하여 가로(cols)와 세로(rows)를 순회합니다. 이때 중요한 것은 push(), translate(), pop()을 사용하는 것입니다. 각 칸(Cell)의 시작점을 (0, 0)으로 기준 잡으면, 칸 내부의 도형을 그릴 때 복잡한 좌표 계산 없이 직관적으로 그릴 수 있습니다. 

function setup() {
  createCanvas(400, 400);
  rectMode(CENTER); // 사각형을 중심 기준으로 그리기
  
  let cols = 10;
  let rows = 10;
  let w = width / cols; // 칸 하나의 너비
  let h = height / rows; // 칸 하나의 높이
  
  for (let i = 0; i < cols; i++) {
    for (let j = 0; j < rows; j++) {
      // 현재 칸의 중심 좌표 계산
      let x = i * w + w/2;
      let y = j * h + h/2;
      
      push(); // 1. 현재 설정 저장
      translate(x, y); // 2. 기준점을 칸의 중심으로 이동
      
      // 모듈로 연산(%)을 이용해 체크무늬 패턴 로직
      if ((i + j) % 2 === 0) {
        fill(255);
        // 칸 크기의 80% 만큼 사각형 그리기 (기준점이 중심이므로 0, 0)
        rect(0, 0, w * 0.8, h * 0.8);
      } else {
        stroke(255);
        noFill();
        circle(0, 0, w * 0.5);
      }
      
      pop(); // 3. 설정 복구 (다음 칸을 위해 좌표계를 원래대로 되돌림)
    }
  }
}
04

노이즈와 벡터 필드

4-1. Perlin Noise의 연속성

random()은 주사위처럼 예측 불가능하게 튀는 값을 주지만, noise()(Perlin Noise)는 구름이나 산맥처럼 부드럽게 이어지는 연속적인 값을 반환합니다. 이 성질을 이용해 화면 전체의 픽셀마다 각도를 부여하면, 물이 흐르는 듯한 Flow Field를 만들 수 있습니다. 입력값(x, y 좌표)에 작은 배율(scale)을 곱해서 노이즈 함수에 넣는 것이 패턴의 핵심입니다. 

let scale = 0.02; // 노이즈의 '줌 레벨' (작을수록 완만하게 변화)

function setup() {
  createCanvas(400, 400);
  background(0);
  stroke(255, 100);
  
  // 그리드 격자마다 노이즈 값을 확인
  for (let x = 0; x < width; x += 15) {
    for (let y = 0; y < height; y += 15) {
      // 2차원 좌표를 노이즈 함수에 전달 (0.0 ~ 1.0 사이 값 반환)
      let n = noise(x * scale, y * scale);
      
      // 노이즈 값을 각도(0 ~ 720도)로 변환
      let angle = n * TWO_PI * 2;
      
      // 해당 각도를 가리키는 벡터 생성
      let v = p5.Vector.fromAngle(angle);
      v.setMag(10); // 선의 길이
      
      push();
      translate(x, y);
      line(0, 0, v.x, v.y); // 흐름의 방향 그리기
      pop();
    }
  }
}

4-2. 노이즈 필드 위를 떠다니는 파티클

이제 벡터 필드를 눈에 보이지 않게 하고, 그 힘을 따라 움직이는 파티클을 띄워봅시다. 각 파티클은 자신의 현재 위치(x, y)에 해당하는 노이즈 값을 읽어와서 이동 방향을 결정합니다. 이 기법은 반 고흐의 "별이 빛나는 밤" 같은 유려한 텍스처를 만듭니다. 

let particles = [];
const scale = 0.01;

function setup() {
  createCanvas(400, 400);
  background(0);
  
  // 100개의 파티클을 랜덤 위치에 생성
  for(let i=0; i<100; i++) {
    particles.push(createVector(random(width), random(height)));
  }
}

function draw() {
  // 궤적을 남기기 위해 아주 투명한 배경을 덮음
  noStroke();
  fill(0, 5); 
  rect(0, 0, width, height);
  
  fill(255);
  
  for(let p of particles) {
    // 1. 현재 위치에서 노이즈 값(방향) 계산
    let n = noise(p.x * scale, p.y * scale);
    let angle = n * TWO_PI * 4; // 각도 증폭
    
    // 2. 해당 각도로 이동하는 힘 벡터 생성
    let force = p5.Vector.fromAngle(angle);
    p.add(force); // 위치 업데이트
    
    // 3. 화면 밖으로 나가면 반대쪽으로 돌아오기 (Wrap around)
    if(p.x < 0) p.x = width;
    if(p.x > width) p.x = 0;
    if(p.y < 0) p.y = height;
    if(p.y > height) p.y = 0;
    
    // 4. 그리기
    circle(p.x, p.y, 2);
  }
}
05

물리법칙과 운동

5-1. 속도와 가속도 (Velocity & Acceleration)

자연스러운 움직임의 핵심은 "위치를 직접 바꾸지 않는 것"입니다. 가속도가 속도를 바꾸고, 속도가 위치를 바꿉니다. 마우스 클릭 시 가속도를 추가하여 속도를 높이고, 마찰력(friction)을 곱해 서서히 멈추게 하는 기본 패턴입니다. 

let pos, vel, acc;

function setup() {
  createCanvas(400, 400);
  pos = createVector(width/2, height/2);
  vel = createVector(0, 0);
  acc = createVector(0, 0);
}

function draw() {
  background(0);
  
  // 물리 연산
  vel.add(acc);
  vel.mult(0.98); // 마찰력 (속도가 점점 줄어듦)
  pos.add(vel);
  acc.mult(0); // 가속도는 매 프레임 초기화
  
  circle(pos.x, pos.y, 30);
}

function mousePressed() {
  // 클릭 시 랜덤한 방향으로 힘(가속도)을 가함
  let force = p5.Vector.random2D();
  force.mult(5);
  acc.add(force);
}

5-2. 충돌과 반사, 거리 계산

물체가 벽에 닿았을 때 속도의 방향을 반대로(`vel.x *= -1`) 뒤집으면 튕겨 나가는 효과를 낼 수 있습니다. 객체 간의 충돌은 dist(x1, y1, x2, y2) 함수로 거리를 계산하여, 두 물체의 반지름 합보다 거리가 가까워졌을 때를 감지합니다. 

let x = 200, y = 200;
let vx = 3, vy = 2;
let r = 20;

function draw() {
  background(0);
  
  x += vx;
  y += vy;
  
  // 벽 충돌 감지
  if (x > width - r || x < r) {
    vx *= -1; // X축 속도 반전
  }
  if (y > height - r || y < r) {
    vy *= -1; // Y축 속도 반전
  }
  
  // 마우스와의 거리 계산 (간단한 인터랙션)
  let d = dist(x, y, mouseX, mouseY);
  if (d < r + 30) { 
    fill(255, 0, 0); // 충돌 시 빨간색
  } else {
    fill(255);
  }
  
  circle(x, y, r * 2);
  
  // 마우스 커서 (장애물 역할)
  noFill();
  stroke(255);
  circle(mouseX, mouseY, 60);
}

5-3. 인력, 척력, 그리고 스프링 (Attraction & Spring)

인력(Attraction)은 목표 지점에서 현재 위치를 뺀 벡터(`target - pos`)를 힘으로 사용합니다. 스프링(Spring)은 훅의 법칙(Hooke's Law)을 따르며, 늘어난 거리만큼 강한 힘으로 돌아오려 하고 거기에 댐핑(Damping)을 주어 진동을 서서히 멈추게 합니다. 탄력 있는 인터랙션의 핵심입니다. 

let pos, vel; // 물체
let restLength = 100; // 스프링 기본 길이
let k = 0.1; // 스프링 강도 (탄성 계수)
let damping = 0.9; // 저항 (감쇠)

function setup() {
  createCanvas(400, 400);
  pos = createVector(200, 300);
  vel = createVector(0, 0);
}

function draw() {
  background(0);
  
  // 앵커 포인트 (마우스)
  let anchor = createVector(mouseX, mouseY);
  
  // 1. 스프링 힘 계산
  let force = p5.Vector.sub(pos, anchor); // 방향 벡터 (물체 -> 앵커)
  let currentLen = force.mag(); // 현재 길이
  let stretch = currentLen - restLength; // 늘어난 길이 (변위)
  
  // 힘의 방향과 크기 설정 (F = -k * x)
  force.normalize();
  force.mult(-1 * k * stretch);
  
  // 2. 물리 적용
  // 중력 추가 (아래로 당기는 힘)
  let gravity = createVector(0, 0.5);
  
  vel.add(force);
  vel.add(gravity);
  vel.mult(damping); // 속도 감쇠
  pos.add(vel);
  
  // 그리기
  stroke(255);
  line(anchor.x, anchor.y, pos.x, pos.y); // 스프링 줄
  fill(255);
  circle(pos.x, pos.y, 40); // 물체
  circle(anchor.x, anchor.y, 10); // 손잡이
}
06

파티클 시스템과 에이전트

6-1. 파티클 시스템 관리 (Lifecycle)

단일 물체가 아니라 수백 개의 파티클을 다룰 때는 배열(Array)을 사용합니다. 중요한 것은 "생명 주기"입니다. 화면 밖으로 나가거나 수명이 다한(Alpha < 0) 파티클을 배열에서 제거(splice)해야 메모리 누수 없이 무한히 솟아나는 효과를 만들 수 있습니다. 

let particles = [];

function draw() {
  background(0);
  
  // 매 프레임 5개의 새로운 파티클 생성
  for(let i=0; i<5; i++) {
    particles.push(new Particle(mouseX, mouseY));
  }
  
  // 배열을 거꾸로 순회해야 삭제 시 인덱스 오류가 안 생김
  for (let i = particles.length - 1; i >= 0; i--) {
    let p = particles[i];
    p.update();
    p.show();
    
    // 수명이 다하면 삭제
    if (p.isDead()) {
      particles.splice(i, 1);
    }
  }
}

class Particle {
  constructor(x, y) {
    this.pos = createVector(x, y);
    this.vel = p5.Vector.random2D().mult(random(1, 3)); // 랜덤 확산
    this.lifespan = 255; // 수명
  }
  
  update() {
    this.pos.add(this.vel);
    this.lifespan -= 5; // 점점 사라짐
  }
  
  show() {
    noStroke();
    fill(255, this.lifespan);
    circle(this.pos.x, this.pos.y, 10);
  }
  
  isDead() {
    return this.lifespan < 0;
  }
}

6-2. 자율 에이전트 (Autonomous Agents)

에이전트는 단순히 물리 법칙에 따르는 것을 넘어, 스스로의 "의지"를 가집니다. Craig Reynolds의 Steering Behaviors 알고리즘을 사용하면, 목표를 추적(Seek)하거나, 장애물을 피하거나, 무리(Flocking)를 짓는 복잡한 생명체의 움직임을 구현할 수 있습니다. 핵심 공식: Steering Force = Desired Velocity - Current Velocity 

class Agent {
  constructor(x, y) {
    this.pos = createVector(x, y);
    this.vel = createVector(0, 0);
    this.acc = createVector(0, 0);
    this.maxSpeed = 6;
    this.maxForce = 0.2; // 조향력 제한 (낮을수록 선회 반경이 큼)
  }

  // 목표 추적 행동 (Seek)
  seek(target) {
    let desired = p5.Vector.sub(target, this.pos);
    
    // 도착 지점에 가까워지면 속도 줄이기 (Arrive Behavior)
    let d = desired.mag();
    if (d < 100) {
      let m = map(d, 0, 100, 0, this.maxSpeed);
      desired.setMag(m);
    } else {
      desired.setMag(this.maxSpeed);
    }
    
    // 조향력 공식
    let steer = p5.Vector.sub(desired, this.vel);
    steer.limit(this.maxForce);
    
    this.acc.add(steer);
  }

  update() {
    this.vel.add(this.acc);
    this.vel.limit(this.maxSpeed);
    this.pos.add(this.vel);
    this.acc.mult(0);
  }

  show() {
    // 진행 방향으로 회전해서 그리기
    let theta = this.vel.heading() + PI/2;
    push();
    translate(this.pos.x, this.pos.y);
    rotate(theta);
    fill(255);
    noStroke();
    triangle(0, -10, -5, 10, 5, 10); // 화살표 모양
    pop();
  }
}

이미지와 픽셀 조작

7-1. 픽셀 샘플링 (Pointillism Effect)

이미지를 있는 그대로 보여주는 것이 아니라, 색상 데이터의 원천으로 활용하는 패턴입니다. img.get(x, y) 함수로 특정 좌표의 색상을 추출한 뒤, 그 색으로 원이나 도형을 그리면 점묘화(Pointillism) 효과를 낼 수 있습니다. 이때 원의 크기를 색상의 밝기(brightness())에 비례하게 하면 더욱 입체적인 표현이 가능합니다. 

let img;

function preload() {
  // 이미지는 무거우므로 preload에서 미리 불러옵니다.
  img = loadImage('assets/sample.jpg'); 
}

function setup() {
  createCanvas(400, 400);
  img.resize(width, height); // 이미지를 캔버스 크기에 맞춤
  background(0);
  noStroke();
}

function draw() {
  // 한 번에 다 그리지 않고, 매 프레임 20개씩 점을 찍어 서서히 그림이 완성되게 함
  for (let i = 0; i < 20; i++) {
    // 1. 랜덤 좌표 선택
    let x = floor(random(width));
    let y = floor(random(height));
    
    // 2. 원본 이미지에서 해당 좌표의 색상 추출
    let col = img.get(x, y);
    
    // 3. 밝기 계산 (0~255)
    let b = brightness(col);
    
    // 4. 밝을수록 큰 원, 어두울수록 작은 원
    let dia = map(b, 0, 255, 2, 15);
    
    fill(col);
    circle(x, y, dia);
  }
}

키네틱 타이포그래피

8-1. 텍스트를 데이터로 변환 (textToPoints)

글자를 이미지로 취급하지 않고, 글자를 구성하는 외곽선 점들의 좌표 배열로 변환하여 다루는 고급 기법입니다. font.textToPoints() 함수를 사용하면 글자의 형상을 유지하면서도 각 점(Point)들을 파티클처럼 흩뿌리거나 흔드는 등 자유로운 변형이 가능해집니다. 

let font;
let points = [];

function preload() {
  font = loadFont('assets/Roboto-Bold.ttf');
}

function setup() {
  createCanvas(400, 200);
  
  // 텍스트를 점들의 좌표 배열로 변환
  // sampleFactor: 점의 밀도 (높을수록 정밀하지만 느려짐)
  points = font.textToPoints('ART', 50, 150, 150, {
    sampleFactor: 0.1
  });
}

function draw() {
  background(0);
  noStroke();
  fill(255, 0, 100);

  for (let p of points) {
    // 각 점의 원래 좌표(p.x, p.y)를 기준으로 노이즈를 더해 흔들림 효과 구현
    // frameCount를 더해 시간이 지남에 따라 노이즈가 흐르게 함
    let n = noise(p.x * 0.01, p.y * 0.01, frameCount * 0.05);
    
    // 노이즈 값(0~1)을 -5~5 픽셀 이동값으로 변환
    let xOffset = map(n, 0, 1, -5, 5);
    let yOffset = map(n, 0, 1, -5, 5);
    
    circle(p.x + xOffset, p.y + yOffset, 3);
  }
}
09

수학과 알고리즘

9-1. 삼각함수 (Oscillation Principles)

크리에이티브 코딩에서 sin()과 cos()은 단순히 삼각형 변을 구하는 도구가 아닙니다. 이 함수들은 입력값(시간, 각도)이 계속 증가하더라도 결과값은 항상 -1과 1 사이를 부드럽게 오가는 파동(Wave)을 만들어냅니다. 이를 map() 함수와 결합하면 투명도, 크기, 위치, 회전 등 모든 속성에 "숨쉬는 듯한" 주기적인 리듬감을 부여할 수 있습니다. 

function draw() {
  background(0);
  translate(width/2, height/2);

  // 시간 변수 (프레임마다 증가)
  let t = frameCount * 0.05;

  // 1. 크기의 진동: sin(t)는 -1~1 -> map을 통해 50~200 픽셀로 변환
  let r = map(sin(t), -1, 1, 50, 200);
  
  // 2. 색상의 진동: cos(t)를 이용해 엇박자로 색상 변화
  let g = map(cos(t), -1, 1, 100, 255);

  fill(255, g, 150);
  circle(0, 0, r);
}

9-2. Lerp (선형 보간법)

lerp(current, target, amount) 함수는 "현재 값"에서 "목표 값"으로 "비율(amount)"만큼 이동한 값을 반환합니다. 예를 들어 비율이 0.1(10%)이라면, 매 프레임마다 남은 거리의 10%씩만 이동하므로 처음에는 빠르게 출발했다가 목표에 가까워질수록 아주 부드럽게 감속하며 도착하는(Ease-out) 효과를 냅니다. 딱딱한 기계적 움직임을 유기적인 움직임으로 바꾸는 마법의 함수입니다. 

let x = 0;
let y = 0;

function draw() {
  background(0, 20); // 잔상을 남기기 위해 투명도 있는 배경
  
  // 마우스 위치(목표)로 매 프레임 5%씩 접근
  // 거리가 멀면 많이 이동하고, 가까우면 조금 이동함 -> 자연스러운 감속
  x = lerp(x, mouseX, 0.05);
  y = lerp(y, mouseY, 0.05);
  
  fill(255);
  noStroke();
  circle(x, y, 30);
}

인터랙션 심화

10-1. 다양한 마우스 이벤트 활용

단순히 `mouseX`, `mouseY`를 읽는 것을 넘어, 사용자의 행동 시점에 따라 정확히 반응해야 합니다. mouseMoved()(단순 이동), mouseDragged()(클릭 후 이동), mousePressed()(누르는 순간), mouseReleased()(떼는 순간) 각각의 이벤트를 조합하여 드로잉 툴이나 인터랙티브 버튼을 만들 수 있습니다. 

function setup() {
  createCanvas(400, 400);
  background(20);
  noStroke();
}

function draw() {
  // draw 루프는 비워두거나 UI 그리는 용도로 사용
  fill(255);
  text("드래그: 그리기 / 클릭: 원형 장식 / 키보드: 지우기", 10, 20);
}

// 1. 마우스를 누른 채 움직일 때 -> 선 그리기
function mouseDragged() {
  stroke(255, 100);
  strokeWeight(5);
  // pmouseX, pmouseY는 바로 직전 프레임의 마우스 좌표
  line(pmouseX, pmouseY, mouseX, mouseY);
}

// 2. 마우스를 꾹 눌렀을 때 -> 포인트 장식
function mousePressed() {
  noStroke();
  fill(255, 50, 100);
  circle(mouseX, mouseY, 20);
}

// 3. 마우스를 뗐을 때 -> 마무리 장식
function mouseReleased() {
  noStroke();
  fill(50, 100, 255);
  circle(mouseX, mouseY, 10);
}

function keyPressed() {
  background(20); // 화면 초기화
}

10-2. 상태 머신 (State Machine) 패턴

복잡한 인터랙티브 아트워크나 게임을 만들 때는 "지금 어떤 장면인가?"를 관리하는 상태 변수(State Variable)가 필수입니다. `switch` 문이나 `if-else` 문을 사용하여 상태에 따라 전혀 다른 `draw` 함수를 실행함으로써, 하나의 캔버스에서 인트로, 메인 인터랙션, 엔딩 화면 등을 전환할 수 있습니다. 

let state = "INTRO"; // 가능한 상태: "INTRO", "PLAY", "GAME_OVER"

function draw() {
  background(0);
  
  // 상태에 따라 다른 화면 그리기
  if (state === "INTRO") {
    drawIntro();
  } else if (state === "PLAY") {
    drawPlay();
  } else if (state === "GAME_OVER") {
    drawGameOver();
  }
}

function drawIntro() {
  fill(255);
  textAlign(CENTER);
  textSize(30);
  text("CLICK TO START", width/2, height/2);
}

function drawPlay() {
  fill(0, 255, 0);
  circle(mouseX, mouseY, 50);
  text("PLAYING...", width/2, 50);
}

function drawGameOver() {
  fill(255, 0, 0);
  textAlign(CENTER);
  text("GAME OVER (Press R)", width/2, height/2);
}

function mousePressed() {
  // 인트로에서 클릭하면 플레이 상태로 전환
  if (state === "INTRO") {
    state = "PLAY";
  } else if (state === "PLAY") {
    // 플레이 중 클릭하면 게임오버로 (예시)
    state = "GAME_OVER";
  }
}

function keyPressed() {
  if (state === "GAME_OVER" && key === 'r') {
    state = "INTRO";
  }
}
11

효과와 필터

그림을 그리는 것(Drawing)만큼이나 이미 그려진 픽셀을 조작하는 것(Post-processing)도 중요합니다. 빛을 표현하거나 잔상을 남기는 등 시각적 완성도를 높이는 기법들입니다.

11-1. 블렌드 모드 (빛의 합성)

기본적으로 p5는 물감을 덮어쓰듯 그림을 그립니다(BLEND 모드). 하지만 blendMode(ADD)를 사용하면 빛을 섞는 것처럼 동작합니다. 색이 겹칠수록 점점 밝아져서 흰색이 되며, 이를 통해 네온 사인, 불꽃, 레이저 같은 발광체 효과(Glow)를 극적으로 표현할 수 있습니다. 

function draw() {
  background(0);
  
  // 1. 빛 효과 켜기
  blendMode(ADD); 
  
  noStroke();
  let t = frameCount * 0.02;
  
  // RGB 색상의 원들을 겹치게 그림
  fill(255, 0, 0);
  circle(width/2 - 20, height/2, 100); // 빨강
  
  fill(0, 255, 0);
  circle(width/2 + 20, height/2 - 30, 100); // 초록
  
  fill(0, 0, 255);
  circle(width/2 + 20, height/2 + 30, 100); // 파랑
  
  // 결과: 세 원이 겹치는 중앙은 빛의 삼원색 원리에 의해 흰색으로 빛남
  
  // 2. 중요: 다음 프레임이나 다른 요소를 위해 모드 초기화
  blendMode(BLEND); 
}

11-2. 잔상 효과 (Alpha Feedback)

가장 간단하면서도 효과적인 트릭입니다. 매 프레임 background(0)으로 화면을 완전히 지우는 대신, 투명도가 있는 배경 background(0, 10)을 덮어씌웁니다. 이렇게 하면 이전 프레임의 그림이 희미하게 남으면서 움직임의 궤적(Trail)이 자연스럽게 생성됩니다. 

function setup() {
  createCanvas(400, 400);
  background(0); // 처음에만 완전 검은색
}

function draw() {
  // 매 프레임 반투명한 검은색 막을 한 겹 씌움
  // 알파값(10)이 낮을수록 궤적이 길게 남음
  background(0, 10); 
  
  fill(0, 255, 255);
  noStroke();
  
  let x = width/2 + cos(frameCount*0.1) * 100;
  let y = height/2 + sin(frameCount*0.1) * 100;
  
  circle(x, y, 20);
}

11-3. 전체 필터 (Global Filters)

filter() 함수는 캔버스 전체의 픽셀을 한 번에 변형합니다. BLUR(흐림), THRESHOLD(임계값 흑백), INVERT(색반전) 등이 자주 쓰입니다. 다만 연산량이 많아 속도가 느려질 수 있으므로 주의해서 사용해야 합니다. 

function draw() {
  // 랜덤한 도형 그리기
  fill(random(255), random(255), random(255));
  circle(random(width), random(height), random(10, 50));
  
  // 일정 주기마다 필터 적용
  if (frameCount % 60 === 0) {
    filter(INVERT); // 화면 색상 반전
  }
  
  if (frameCount % 120 === 0) {
    filter(BLUR, 3); // 화면 뭉개기 (몽환적 효과)
  }
}
12

모듈화 (Instance Mode)

12-1. 전역 오염 방지 (Namespace)

일반적인 p5.js 방식(Global Mode)은 변수들이 전역 공간에 노출되어, React/Vue 같은 프레임워크나 여러 개의 스케치를 한 페이지에 띄울 때 충돌이 발생합니다. Instance Mode는 p라는 객체 안에 p5의 모든 기능(setup, draw, circle 등)을 가두어 두는 패턴입니다. 이를 통해 코드를 안전하게 모듈화하고 재사용할 수 있습니다. 

import p5 from 'p5';

// 스케치를 하나의 함수로 감쌉니다. (Closure)
const sketch = (p) => {
  let x = 0; // 이 변수는 이 스케치 안에서만 유효합니다.

  p.setup = () => {
    // 모든 p5 함수 앞에 'p.'을 붙여야 합니다.
    p.createCanvas(400, 400);
  };

  p.draw = () => {
    p.background(0);
    p.fill(255);
    p.circle(x, p.height/2, 50);
    x++;
  };
};

// 새로운 p5 인스턴스 생성 (특정 div에 붙일 수도 있음)
// new p5(sketch, document.getElementById('canvas-container'));
new p5(sketch);
